\usepackage{xcolor}
\usepackage{listings}

% Colors
\definecolor{codebg}{RGB}{245,245,245}
\definecolor{codeframe}{RGB}{220,220,220}

% --------------------------
% Python
% --------------------------
\lstdefinelanguage{Python}{
  keywords={def, return, if, else, elif, for, while, import, as, from, class, try, except, finally, with, pass, break, continue, lambda, yield, raise, assert, True, False, None},
  keywordstyle=\color{purple}\bfseries,
  comment=[l]{\#},
  commentstyle=\color{gray}\itshape,
  string=[s]{"}{"},         % double-quoted
  morestring=[s]{'}{'},     % single-quoted
  morestring=[s]{""" }{ """}, % triple double-quoted
  morestring=[s]{'''}{'''},   % triple single-quoted
  stringstyle=\color{orange!70!black},
  sensitive=true
}

% --------------------------
% Bash
% --------------------------
\lstdefinelanguage{Bash}{
  keywords={if, then, else, elif, fi, for, while, do, done, function, return, exit, case, esac, in},
  keywordstyle=\color{purple}\bfseries,
  comment=[l]{\#},
  commentstyle=\color{gray}\itshape,
  string=[b]",
  morestring=[b]',
  stringstyle=\color{orange!70!black},
  sensitive=true
}

% --------------------------
% SQL
% --------------------------
\lstdefinelanguage{SQL}{
  keywords={SELECT, INSERT, UPDATE, DELETE, FROM, WHERE, JOIN, ON, GROUP, BY, ORDER, ASC, DESC, CREATE, TABLE, DROP, ALTER, INTO, VALUES, SET, DISTINCT},
  keywordstyle=\color{red}\bfseries,
  comment=[l]{--},
  commentstyle=\color{gray}\itshape,
  string=[b]',
  stringstyle=\color{green!50!black},
  sensitive=false
}

% --------------------------
% General layout (all languages)
% --------------------------
\lstset{
  basicstyle=\small\ttfamily,
  basewidth=0.5em,
  breaklines=true,
  breakatwhitespace=true,
  backgroundcolor=\color{codebg},
  frame=single,
  rulecolor=\color{codeframe},
  frameround=tttt,
  framesep=5pt,
  showstringspaces=false,
  tabsize=2
}
